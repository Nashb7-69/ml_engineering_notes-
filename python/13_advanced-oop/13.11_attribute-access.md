# 13.11 Attribute Access Customization: __getattribute__, __getattr__, __setattr__, __delattr__, __dir__

## Two Entry Points for Access: Reliable vs Fallback

Python provides two interception hooks for reading attributes, and they are *not* interchangeable:

| Hook | Invoked when | Typical use |
|------|-------------|-------------|
| `__getattribute__(self, name)` | On **every** `obj.name`, `getattr(obj, name)`, `obj.__dict__["name"]` bypass | Total control, proxies, lazy wiring, access logging |
| `__getattr__(self, name)` | **Only if** normal lookup fails (after `__getattribute__` + MRO + `__dict__` yielded nothing) | Virtual / dynamic attributes, delegation, computed fallback, typo suggestions |

```python
class Demo:
    def __getattribute__(self, name):
        print(f"__getattribute__ {name!r}")
        return object.__getattribute__(self, name)

    def __getattr__(self, name):
        print(f"__getattr__ fallback for {name!r}")
        return f"<missing {name}>"

d = Demo()
d.x = 10
print(d.x)            # __getattribute__ 'x' -> 10 — __getattr__ not called
print(d.missing)      # __getattribute__ 'missing' -> AttributeError -> __getattr__ 'missing' -> "<missing missing>"
print(d.__dict__)     # __getattribute__ '__dict__' -> {x:10} — __getattribute__ even fires for __dict__ itself!
```

> **Mental model:** `__getattribute__` **replaces** the primary lookup algorithm. `__getattr__` is the *last resort* called only when that algorithm declared "not found". Think of `__getattribute__` as "I will decide for every attribute" and `__getattr__` as "I only care about the misses".

### The Precise Lookup Order For `obj.attr`

1.  Call `type(obj).__getattribute__(obj, name)` if defined; otherwise use default (`object.__getattribute__`), which does:
    a.  Search *data descriptors* on the type and its bases (objects with `__set__`).
    b.  Search `obj.__dict__` (or slots) for `name`.
    c.  Search *non-data descriptors* and plain class attributes; bind via `__get__` if descriptor.
    d.  If none of the above found anything, raise `AttributeError`.
2.  If step 1 raised `AttributeError`, look for `__getattr__` on the type and, if present, call `type(obj).__getattr__(obj, name)`.
3.  If that still raises/returns `AttributeError` not rescue, raise `AttributeError`.

Hence, **data descriptors shadow `__dict__`, but `__getattr__` only fires after all those slots have missed.** This interacts directly with lesson **13.10**: descriptors live inside `__getattribute__` machinery.

## `__getattribute__` — Universal Interception (With Great Power, Great Recursion Risk)

### The Infinite Recursion Trap

The moment you override `__getattribute__`, even innocuous reads like `self.x` inside it will themselves call `__getattribute__`:

```python
class Broken:
    def __getattribute__(self, name):
        # ❌ Infinite recursion — getattr(self, name) calls __getattribute__ again!
        # return getattr(self, name)  # RecursionError

        # ❌ Also recursion — self.__dict__[name] still needs attribute access for __dict__ itself
        # return self.__dict__[name]

        # ✅ Correct delegation — bypasses __getattribute__ by calling the base implementation
        return object.__getattribute__(self, name)

# Similarly: `self.attr` inside __getattribute__ triggers __getattribute__ recursively.
# So inside __getattribute__, NEVER do `self.X` or `getattr(self, X)` unless you delegate via
# object.__getattribute__(self, X) or object.__setattr__(self, X).
```

### Example 1: Access Logger

```python
class LoggedAccess:
    """Logs every attribute read — useful for debugging attribute storms."""
    def __init__(self, value):
        # Important: initialize _value WITHOUT triggering overridden __setattr__
        # We'll write __setattr__ later — if it logs too, assigning in __init__ would log.
        object.__setattr__(self, "value", value)

    def __getattribute__(self, name):
        # Fetch via object.__getattribute__ to avoid self-referential lookup of these names
        # Don't spam logs for dunder / private during demo:
        # Use get() style guard with object.__getattribute__ for logging switch
        if name in ("value", "__dict__", "__class__"):
            return object.__getattribute__(self, name)
        print(f"[get] {name!r}")
        return object.__getattribute__(self, name)

    def __repr__(self): return f"LoggedAccess(value={self.value!r})"

a = LoggedAccess(42)
print(a.value)         # 42 — no log for value (excluded path)
print(a.__class__)     # LoggedAccess — no log
# Adding arbitrary attribute triggers log:
# print(a.nonexistent) # [get] 'nonexistent' -> AttributeError

# Edge: even log assignment `print` lookups etc. may fire __getattribute__ — keep it minimal and fast.
# Hot path cost: every attribute access now pays a Python call; 10–50× slower than default.
# Only use for proxies, security sandboxes, or temporary debugging — never as a default interception on all domain types.
```

### Example 2: Read-Only Proxy (Without Data Descriptor)

```python
class ReadOnlyProxy:
    """Proxy that forbids attribute writes at all — uses __getattribute__ for reads, __setattr__ blocks writes."""
    def __init__(self, wrapped):
        object.__setattr__(self, "_wrapped", wrapped)

    def __getattribute__(self, name):
        # Bypass proxy for its own bookkeeping attributes
        if name in ("_wrapped", "__dict__", "__class__", "__slots__"):
            return object.__getattribute__(self, name)
        wrapped = object.__getattribute__(self, "_wrapped")
        return getattr(wrapped, name)  # delegates to wrapped — getattr's lookup is on wrapped's type, not proxy's

    def __setattr__(self, name, value):
        if name == "_wrapped":
            object.__setattr__(self, name, value)
            return
        raise AttributeError(f"{type(self).__name__} is read-only: cannot set {name!r}")

    def __delattr__(self, name):
        raise AttributeError(f"{type(self).__name__} is read-only: cannot delete {name!r}")

    def __repr__(self): return f"ReadOnlyProxy({object.__getattribute__(self, '_wrapped')!r})"

class Counter:
    def __init__(self): self.count = 0
    def inc(self): self.count += 1
    def __repr__(self): return f"Counter(count={self.count})"

c = Counter()
p = ReadOnlyProxy(c)
print(p.count)         # 0 — proxied
p.inc()                # mutates wrapped's private state via method — still allowed
print(p.count)         # 1 — observes mutation
try: p.count = 99
except AttributeError as e: print(e)
try: p.new = 5
except AttributeError as e: print(e)
# Note: mutable methods on wrapped can still change state — proxy is read-only for *attribute writes*, not for
# logical immutability. To guarantee immutability, wrap methods too or copy.
```

## `__getattr__` — Fallback for Dynamic / Delegated Attributes

Because it only fires on miss, `__getattr__` carries zero overhead for normal reads. It is the right tool for proxies, lazy virtual attributes, and computed fallback.

### Example 1: Delegation Proxy (Compose Without Inheritance)

```python
class AutoDelegator:
    """Object that exposes a dict-backed API under a prefix — virtual attributes without descriptors."""
    def __init__(self, data: dict, prefix: str = "cfg_"):
        self._data, self._prefix = dict(data), prefix

    def __getattr__(self, name):
        # Only called when normal lookup missed — so explicit methods/attrs take priority
        if name.startswith(self._prefix):
            key = name[len(self._prefix):]
            try:
                return self._data[key]
            except KeyError:
                raise AttributeError(f"{name!r} (no key {key!r} in backing data)") from None
        raise AttributeError(f"{type(self).__name__!r} has no attribute {name!r}")

    def __dir__(self):
        # Support dir() and autocomplete: expose virtual attrs
        extra = [self._prefix + k for k in self._data]
        return sorted(set(object.__dir__(self)) | set(extra))

d = AutoDelegator({"host": "localhost", "port": 8080})
print(d.cfg_host)       # localhost — via __getattr__
print(d.cfg_port)       # 8080
try: print(d.cfg_missing)
except AttributeError as e: print(e)
print(d._data)          # direct attribute — found via normal lookup, __getattr__ not called
print("cfg_host" in dir(d))  # True — __dir__ advertised virtual attrs

# Important: __getattr__ must raise AttributeError for truly missing names.
# If it returns a fallback value for every name, `hasattr(obj, name)` always reports True
# and typo errors become silent.
```

### Example 2: Lazy Computation Cache via `__getattr__`

```python
import math

class Circle:
    """Radius known at init; diameter, circumference, area computed lazily and then cached as instance attributes."""
    def __init__(self, radius: float):
        self.radius = float(radius)

    def __getattr__(self, name):
        # Called only when attribute not found — signals we need to compute a derived attribute
        if name == "diameter":
            value = self.radius * 2
        elif name == "area":
            print("  computing area…")   # demo that it's lazy
            value = math.pi * self.radius ** 2
        elif name == "circumference":
            print("  computing circumference…")
            value = 2 * math.pi * self.radius
        else:
            raise AttributeError(f"{type(self).__name__!r} has no attribute {name!r}")

        # Cache: store computed value directly in __dict__ so next access is O(1) and skips __getattr__
        setattr(self, name, value)  # uses normal __setattr__; or object.__setattr__(self, name, value) if custom setter
        return value

    def __repr__(self): return f"Circle(radius={self.radius})"

c = Circle(3)
print(c.area)       # computing area… 28.27...
print(c.area)       # 28.27... — cached, no recomputation, no __getattr__ call
print(c.__dict__)   # {'radius': 3.0, 'area': 28.27...} — area now a concrete attribute
c.radius = 5
# Cache is now stale! Must invalidate — either delete c.area so recomputed, or don't cache lazily if dependencies mutable.
del c.area          # invalidate; next access recomputes
print(c.area)       # computing area… 78.5...

# Compare to descriptor `Cached` in 13.10: __getattr__ achieves the same with less boilerplate
# but only for *dynamic* names; Fixed fields should use descriptors.
```

### Example 3: Method-Missing Style — Auto-Generated API Methods

```python
class RPC:
    """Simulates generating client methods per remote endpoint — `client.users()` calls remote users endpoint."""
    def __getattr__(self, name):
        # Avoid generating dunders/private — they reflect class internals, not remote endpoints
        if name.startswith("_"):
            raise AttributeError(name)

        def rpc_method(*args, **kwargs):
            # Real code would HTTP POST to f"/api/{name}"
            return f"RPC:{name}({args},{kwargs})"

        # Optional: cache the generated method on the instance for fast second call
        # setattr(self, name, rpc_method)
        return rpc_method

    def __dir__(self):
        # If endpoints are known statically, advertise them here
        known = ["users", "orders", "health"]
        return sorted(set(known) | set(object.__dir__(self)))

client = RPC()
print(client.users(limit=10))   # RPC:users(((), {'limit': 10})) — auto-generated method
print(client.health())          # RPC:health
print(hasattr(client, "orders")) # False — __getattr__ exists but hasattr triggers __getattr__ via __getattribute__ fallback
# Actually hasattr(client, "orders") WOULD try __getattr__("orders") and get a callable -> True
# But by not pre-caching, `hasattr` still sees it:
print(hasattr(client, "orders"))  # True — because __getattr__("orders") returns a function — truthy; hasattr checks `except AttributeError`
# For `hasattr` correctness, always raise AttributeError for not-found names inside __getattr__.
```

## `__setattr__` / `__delattr__` — Write Interception

These hooks fire for *every* attribute assignment/deletion (`obj.name = value` / `del obj.name`) before `__dict__` write.

```python
class Validated:
    """Reject writes that don't pass validation — extends reads with write-side policy."""
    def __init__(self, x: int):
        # Must use object.__setattr__ inside __init__ if __setattr__ validates — otherwise recursion or re-validation
        object.__setattr__(self, "x", x)
        object.__setattr__(self, "_frozen", False)

    def __getattribute__(self, name):
        # Minimal — just delegate correctly
        return object.__getattribute__(self, name)

    def __setattr__(self, name, value):
        # Check frozen after init:
        try:
            frozen = object.__getattribute__(self, "_frozen")
        except AttributeError:
            frozen = False  # during __init__ before _frozen exists
        if frozen and name != "_frozen":
            raise AttributeError(f"{type(self).__name__} is frozen — cannot set {name!r}")
        if name == "x" and not isinstance(value, int):
            raise TypeError(f"x must be int, got {type(value).__name__}")
        # Bypass any descriptor protocol by writing directly to __dict__ or via object.__setattr__
        object.__setattr__(self, name, value)

    def freeze(self):
        object.__setattr__(self, "_frozen", True)

    def __repr__(self): return f"Validated(x={self.x!r})"

v = Validated(10)
print(v)            # Validated(x=10)
v.x = 20
print(v)
try: v.x = "bad"
except TypeError as e: print(e)
v.freeze()
try: v.x = 30
except AttributeError as e: print(e)
```

## Interaction: Descriptors vs `__getattribute__` / `__getattr__` / `__dict__`

Order recall for `obj.x` **get**:

```
obj.x
  → type(obj).__getattribute__(obj, "x")   # default object.__getattribute__
        1. look for data descriptor on type MRO — if found, call desc.__get__(obj, type)
        2. else check instance.__dict__["x"] — return if present
        3. else look for non-data descriptor / plain class attr — call __get__ if descriptor else return value
        4. else if AttributeError, delegate to type.__getattr__(obj, "x") if defined
        5. else raise AttributeError
```

For `obj.x = v` **set**:

```
obj.x = v
  → type(obj).__setattr__(obj, "x", v)   # if defined (oversees entire set)
        else:
          look for data descriptor on type — if found, call desc.__set__(obj, v)
          else write instance.__dict__["x"] = v
```

```python
from functools import cached_property

# Demo: data descriptor shadows __dict__ — even if __dict__ has an entry, descriptor wins
class HideViaData:
    def __init__(self, v): self._v = v
    @property
    def x(self): return self._v * 2

h = HideViaData(10)
h.__dict__["x"] = 999  # manual insertion
print(h.x)             # 20 — property (data descriptor) wins, 999 ignored
print(h.__dict__["x"]) # 999 — still there, but shadowed
del h.__dict__["x"]
print(h.x)             # 20 — still property

# Non-data descriptor loses to __dict__
class Func:
    def __get__(self, obj, owner): return "from descriptor"
    # no __set__ — non-data

class ND:
    attr = Func()

nd = ND()
print(nd.attr)              # from descriptor
nd.__dict__["attr"] = "from dict"
print(nd.attr)              # from dict — shadowing succeeded
del nd.__dict__["attr"]
print(nd.attr)              # from descriptor — restored

# __getattr__ never fires when descriptor or __dict__ existed, even if value is e.g. None
class G:
    present = 0
    def __getattr__(self, name): return "fallback"

print(G().present)   # 0 — descriptor/class-attr found, __getattr__ not called (even though 0 is falsy)
# Only truly missing causes __getattr__:
print(G().missing)   # fallback
```

## `__dir__`, `__slots__`, and Introspection Compatibility

`__dir__` shapes what `dir(obj)`, tab-completion, and IDE tools advertise. For dynamic attributes, override `__dir__` to include what `__getattr__` can serve:

```python
class Store:
    def __init__(self, data): self._data = dict(data)
    def __getattr__(self, name):
        try: return self._data[name]
        except KeyError: raise AttributeError(f"no attribute {name!r}") from None
    def __dir__(self):
        # Advertise stored keys and normal attributes
        return sorted(set(self._data.keys()) | set(object.__dir__(self)))

s = Store({"x": 1, "y": 2})
print(dir(s))               # includes x, y
print("x" in dir(s))        # True — IDE autocomplete sees x
print(hasattr(s, "x"), getattr(s, "x"))  # True, 1 — __getattr__ provides it
# Note: hasattr calls __getattr__ internally via getattr; if __getattr__ raised AttributeError,
# hasattr returns False — which dir + getattr interplay exploits for discovery.
# For slotted classes, ensure dir accounts for slots even before they're populated:
class Slotted:
    __slots__ = ("a", "b", "__weakref__")
    def __init__(self): self.a = 1
    def __dir__(self):
        # Include slots even if not yet assigned — completeness for callers
        return sorted(set(super().__dir__()) | {"a", "b"})
```

## Advanced: Delegating Proxy That Passes `isinstance` and `__class__`

```python
class IsolatedProxy:
    """Proxy that delegates most attribute operations but preserves its own identity."""
    def __init__(self, wrapped): object.__setattr__(self, "_w", wrapped)

    def __getattr__(self, name):   # only misses
        return getattr(object.__getattribute__(self, "_w"), name)

    def __setattr__(self, name, value):
        if name == "_w": object.__setattr__(self, name, value)
        else:
            setattr(object.__getattribute__(self, "_w"), name, value)

    def __delattr__(self, name):
        if name == "_w": object.__delattr__(self, name)
        else: delattr(object.__getattribute__(self, "_w"), name)

    def __dir__(self): return dir(object.__getattribute__(self, "_w"))

    # Make proxy look like wrapped for isinstance/inspection? Usually don't — preserve identity.
    # But to mimic type, expose __class__:
    @property
    def __class__(self):
        # Overriding __class__ attribute can make `type(proxy)` returns wrapped's type — but
        # `type(obj)` is slot, not normal attribute; this property only affects `obj.__class__`
        return object.__getattribute__(self, "_w").__class__

# Nuance: type(proxy) is still IsolatedProxy — `isinstance(proxy, Wrapped)` remains False.
# To make isinstance lie, use __instancecheck__ on a custom metaclass — rarely desired.
# Prefer to NOT masquerade type for debugging sanity.
```

### `__slots__` Interaction

*   `__getattribute__` is still the lookup entry — slots are just an alternate storage queried after data descriptors and before class attributes. A slotted class still invokes descriptors and then falls through to slots search.
*   `__setattr__` is essential with slots to validate or translate slot names; always delegate via `object.__setattr__` to store into slots.

```python
class S:
    __slots__ = ("x", "y", "__dict__")  # __dict__ alias allows extra dynamic attrs alongside fixed slots
    def __init__(self, x, y): self.x, self.y = x, y
    def __getattribute__(self, name):
        print(f"  S.__getattribute__ {name!r}")
        return object.__getattribute__(self, name)
    def __setattr__(self, name, value):
        print(f"  S.__setattr__ {name!r}={value!r}")
        object.__setattr__(self, name, value)
    def __getattr__(self, name):  # only after slots/__dict__ miss
        return f"<virtual {name}>"

s = S(1, 2)
print(s.x)        # logs both getattribute hits for `x` plus internal lookups
print(s.z)        # <virtual z> — after S.__getattribute__ raised AttributeError for z, __getattr__ handled it
```

## Comparison Checklist: When to Use Which

| Mechanism | When | Overhead per access | Storage |
|-----------|------|---------------------|---------|
| Data descriptor (`Typed`, `Range`) | Fixed, validated, typed, reused attributes | One Python `__get__`/`__set__` call; fast | Private instance dict key or external map |
| `property` | One computed/validated attribute on one class | One function call | Closure/private dict |
| `__getattr__` (fallback only) | Virtual/dynamic, delegation, proxies | Zero for normal hits; one Python call only on misses | Dynamic map, remote fetch, faker |
| `__getattribute__` (all accesses) | Total control, audit, proxies needing read interception | One Python call per access — biggest overhead | Any — but must delegate via `object.__getattribute__` |
| `__setattr__`, `__delattr__` | Validation, freezing, audit on write | One Python call per write | Delegation via `object.__setattr__` |

> **Rule of thumb:** Start with descriptor/property for known names. Add `__getattr__` for dynamic names. Resort to `__getattribute__` only when you must intercept every access and understand the recursion consequences.

## Common Mistakes & Hard Lessons

| Mistake | Symptom | Fix |
|---------|---------|-----|
| `__getattribute__` uses `getattr(self, name)` or `self.__dict__[name]` | `RecursionError: maximum recursion depth exceeded` | `object.__getattribute__(self, name)` / `super().__getattribute__(name)` |
| `__setattr__` uses `self.name = value` | Recurses, `RecursionError` | `object.__setattr__(self, name, value)` / `super().__setattr__(name, value)` |
| `__getattr__` returns fallback for every name | `hasattr(obj, "typoo")` incorrectly `True`; typos silenced | `raise AttributeError(f"{name!r} missing")` for truly missing |
| `__getattr__` handles `name.startswith("_")` but raises wrong exception type | Masks errors; debuggers see `RuntimeError` not `AttributeError` | Raise only `AttributeError` for `__getattr__` not-found (Python spec — `getattr` catches it) |
| Not implementing `__dir__` for dynamic `__getattr__` | Tab complete / `dir()` misses virtual attrs | `def __dir__(self): return sorted(set(dynamic_keys) \| set(super().__dir__()))` |
| Not considering descriptors vs `__getattr__` priority | Virtual `__getattr__` shadowed by data descriptor, surprising | Expected: data descriptors win per spec; document that known validated fields trump virtuals and test |
| Using `__getattribute__` to implement lazy virtual attrs | Penalty on every access, including internals | Use `__getattr__` for lazy; reserve `__getattribute__` for proxies that need per-access logic |

## Realistic Consolidation: Hybrid Validation + Dynamic Store

```python
class ValidatedWithDynamicFields:
    """
    Validated fixed field `price` (descriptor semantics) + arbitrary metadata via `meta_*` fallback.
    Shows that descriptor and __getattr__ coexist per design.
    """
    class _Price:
        def __set_name__(self, owner, name): self.private = f"_{name}"
        def __get__(self, obj, owner=None):
            if obj is None: return self
            return getattr(obj, self.private, 0.0)
        def __set__(self, obj, value):
            if not isinstance(value, (int, float)): raise TypeError("price must be numeric")
            if value < 0: raise ValueError("price must be >= 0")
            setattr(obj, self.private, float(value))

    price = _Price()

    def __init__(self, price, **meta):
        self.price = price           # via descriptor (validates)
        self._meta: dict[str, object] = dict(meta)

    def __getattr__(self, name: str):
        if name.startswith("meta_"):
            key = name[5:]
            try: return self._meta[key]
            except KeyError: raise AttributeError(f"no metadata key {key!r}") from None
        raise AttributeError(f"{type(self).__name__!r} has no attribute {name!r}")

    def __setattr__(self, name, value):
        # Allow descriptor for price, _meta for metadata, else delegate
        if name in ("price", "_meta"):
            # Route price through data descriptor; _meta through normal storage
            # Must delegate correctly — super().__setattr__ respects descriptors
            super().__setattr__(name, value)
        elif name.startswith("meta_"):
            # Dynamic metadata write — virtual attributes
            key = name[5:]
            self._meta[key] = value
        else:
            super().__setattr__(name, value)

    def __dir__(self):
        dynamic = [f"meta_{k}" for k in self._meta]
        return sorted(set(dynamic) | set(super().__dir__()))

obj = ValidatedWithDynamicFields(9.99, color="red")
print(obj.price)       # 9.99 — descriptor, fast path
print(obj.meta_color)  # red — __getattr__ fallback
obj.meta_size = "M"
print(obj._meta)       # {'color': 'red', 'size': 'M'}
try: obj.price = -1
except ValueError as e: print(e)
try: print(obj.meta_unknown)
except AttributeError as e: print(e)
```

## Performance & Design Guidance

*   `__getattribute__` cost: every read pays a Python call. On tight loops (millions of lookups), measured overhead is ~5–20× over raw `__dict__` access. Use only for proxies and instrumentation.
*   `__getattr__` cost: only on cache-miss attribute; if virtual attributes are rare, overhead is near-zero.
*   For virtual-attribute-heavy types (ORM row, config tree), prefer descriptor/indexing (`__getitem__`) when key is not known statically; `obj["key"]` is more explicit than `obj.key_via_getattr`.
*   Design checklist:
    1.  Is name fixed and known at class definition? → descriptor / `property`.
    2.  Is name dynamic, unknown, from external source? → `__getattr__` / `__getitem__`.
    3.  Do you need to intercept even *present* reads (security, audit, proxy delegation of every access)? → `__getattribute__` with recursion-safe delegation and minimal per-call work.

## Advanced Proxy: Lazy Module and Security Patterns

### Lazy Module Proxy

```python
import importlib
class LazyModule:
    def __init__(self, name: str):
        object.__setattr__(self, "_name", name)
        object.__setattr__(self, "_module", None)
    def _load(self):
        mod = object.__getattribute__(self, "_module")
        if mod is None:
            name = object.__getattribute__(self, "_name")
            mod = importlib.import_module(name)
            object.__setattr__(self, "_module", mod)
        return mod
    def __getattr__(self, name):
        return getattr(self._load(), name)
    def __dir__(self):
        mod = object.__getattribute__(self, "_module")
        if mod is None: return object.__dir__(self)
        return sorted(set(dir(mod)) | set(object.__dir__(self)))

lazy_json = LazyModule("json")
print(lazy_json.dumps({"a": 1}))
```

### Security Proxy

```python
class SecureProxy:
    def __init__(self, wrapped):
        object.__setattr__(self, "_wrapped", wrapped)
    def __getattribute__(self, name):
        if name.startswith("_") and name not in ("_wrapped", "__class__"):
            raise AttributeError(f"Access to private {name!r} denied")
        if name == "_wrapped": return object.__getattribute__(self, name)
        wrapped = object.__getattribute__(self, "_wrapped")
        return getattr(wrapped, name)
    def __setattr__(self, name, value):
        if name.startswith("_"): raise AttributeError(f"Cannot set private {name!r}")
        setattr(object.__getattribute__(self, "_wrapped"), name, value)
```
